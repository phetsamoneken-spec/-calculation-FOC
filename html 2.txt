    <script>
        const dataTiers = [
            { min: 2000.010, beerRate: 6100, csdRate: 4800, foc: 8, label: ">= 2,000.010" },
            { min: 1800.010, beerRate: 5800, csdRate: 4500, foc: 7, label: "1,800.010 - 2,000" },
            { min: 1200.010, beerRate: 5500, csdRate: 4200, foc: 6, label: "1,200.010 - 1,800" },
            { min: 800.010, beerRate: 5200, csdRate: 3900, foc: 5, label: "800.010 - 1,200" },
            { min: 400.010, beerRate: 4800, csdRate: 3500, foc: 4, label: "400.010 - 800" },
            { min: 0.020, beerRate: 4300, csdRate: 3000, foc: 3, label: "0.020 - 400" },
            { min: 0, beerRate: 0, csdRate: 0, foc: 2, label: "<= 0.010" }
        ];

        function formatNumber(num, digits = 0) {
            return new Intl.NumberFormat('lo-LA', {
                minimumFractionDigits: digits,
                maximumFractionDigits: digits
            }).format(num);
        }

        function calculate() {
            const beerPrice = parseFloat(document.getElementById('beer-price-input').value) || 1;
            const csdPrice = parseFloat(document.getElementById('csd-price-input').value) || 1;
            
            const beerVol = parseFloat(document.getElementById('beer-volume').value) || 0;
            const csdVol = parseFloat(document.getElementById('csd-volume').value) || 0;
            const totalVol = beerVol + csdVol;
            
            const resultCard = document.getElementById('result-card');
            const totalDisplay = document.getElementById('total-display');
            const emptyState = document.getElementById('empty-state');

            if (totalVol <= 0 && document.getElementById('beer-volume').value === "" && document.getElementById('csd-volume').value === "") {
                resultCard.classList.add('hidden');
                totalDisplay.classList.add('hidden');
                emptyState.classList.remove('hidden');
                return;
            }

            resultCard.classList.remove('hidden');
            totalDisplay.classList.remove('hidden');
            emptyState.classList.add('hidden');

            const tier = dataTiers.find(t => totalVol >= t.min) || dataTiers[dataTiers.length - 1];

            const beerDiscountKip = beerVol * tier.beerRate;
            const csdDiscountKip = csdVol * tier.csdRate;

            const beerDiscountCrate = beerDiscountKip / beerPrice;
            const csdDiscountCrate = csdDiscountKip / csdPrice;

            document.getElementById('total-vol-val').innerText = totalVol.toLocaleString('en-US', {maximumFractionDigits: 3});
            document.getElementById('foc-val').innerText = tier.foc;
            
            document.getElementById('beer-discount-crate').innerText = isFinite(beerDiscountCrate) ? beerDiscountCrate.toFixed(3) : "0.000";
            document.getElementById('beer-discount-kip').innerText = `ມູນຄ່າ: ${formatNumber(Math.round(beerDiscountKip))} ກີບ`;
            
            document.getElementById('csd-discount-crate').innerText = isFinite(csdDiscountCrate) ? csdDiscountCrate.toFixed(3) : "0.000";
            document.getElementById('csd-discount-kip').innerText = `ມູນຄ່າ: ${formatNumber(Math.round(csdDiscountKip))} ກີບ`;
            
            // Show combined bonus summary
            document.getElementById('total-beer-bonus').innerText = (tier.foc + (isFinite(beerDiscountCrate) ? beerDiscountCrate : 0)).toFixed(3);
            document.getElementById('total-water-bonus').innerText = (isFinite(csdDiscountCrate) ? csdDiscountCrate : 0).toFixed(3);
            
            document.getElementById('tier-info').innerText = `ຂັ້ນ: ${tier.label}`;
        }
    </script>
